Python 绿幕抠图算法
# 绿幕精确抠图（基于实际背景色检测）
from PIL import Image
import numpy as np

# 1. 检测实际背景色（AI 生成的绿幕偏黄）
# 采样边缘像素，统计最常见颜色
bg_color = np.array([162, 245, 33])  # 黄绿色，非纯绿

# 2. 计算每个像素与背景色的欧几里得距离
diff = np.abs(data[:,:,:3].astype(float) - bg_color.astype(float))
distance = np.sqrt(np.sum(diff**2, axis=2))

# 3. 创建透明度（距离阈值 60 + 渐变过渡 40）
alpha = np.zeros(distance.shape, dtype=np.uint8)
alpha[distance > 100] = 255  # 完全不透明
mask = (distance > 60) & (distance <= 100)
alpha[mask] = ((distance[mask] - 60) / 40 * 255).astype(np.uint8)

# 4. 应用 Alpha 通道并保存
data[:,:,3] = alpha
Image.fromarray(data).save('bigben_greenscreen_clean.png', 'PNG')
