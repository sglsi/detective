Python 抠图算法
# 基于颜色距离的 Alpha 通道生成
from PIL import Image
import numpy as np

# 1. 自动检测背景色（统计边缘像素）
edge_pixels = data[:10, :].reshape(-1, 3)
bg_color = np.mean(edge_pixels, axis=0)

# 2. 计算每个像素与背景色的欧几里得距离
diff = np.abs(data[:,:,:3].astype(float) - bg_color.astype(float))
distance = np.sqrt(np.sum(diff**2, axis=2))

# 3. 创建透明度（距离阈值 40 + 渐变过渡 30）
alpha = np.zeros(distance.shape, dtype=np.uint8)
alpha[distance > 70] = 255  # 完全不透明
mask = (distance > 40) & (distance <= 70)
alpha[mask] = ((distance[mask] - 40) / 30 * 255).astype(np.uint8)

# 4. 应用 Alpha 通道并保存
data[:,:,3] = alpha
Image.fromarray(data).save('bigben_transparent.png', 'PNG')
